+def _vtl_conv_gate_spec_eligible(attn_metadata, conv_state, state_indices) -> bool:
+    """True if the fused kernel can serve this CHAIN-SPEC decode batch.
+
+    Deliberately NOT cached: unlike the shape predicate, every term here is per-step. A batch
+    that qualifies now may not next step (APC can turn on, query_start_loc_d can be absent), and
+    a cached `True` would keep the kernel running with a rollback it cannot compute.
+
+    Fail-closed, and evaluated BEFORE anything mutates conv_state -- once the ring buffer has
+    advanced, falling back to the stock path would double-apply the update.
+    """
+    if not _VTL_BCX_CONV_GATE_SPEC:
+        return False
+    try:
+        # The kernel derives each request's extent from query_start_loc itself; without it there
+        # is no way to split the flat token axis into requests.
+        if attn_metadata.query_start_loc_d is None:
+            return False
+        # APC / mamba_cache_mode="all": state_indices becomes the full block table and the Triton
+        # path picks the live block via initial_state_idx / block_idx_last_scheduled_token. This
+        # kernel only ever reads column 0, so it would silently address the wrong block.
+        if getattr(attn_metadata, "block_idx_last_scheduled_token", None) is not None:
+            return False
+        if state_indices is None or state_indices.dim() != 2:
+            return False
+        # Every remaining condition MIRRORS a TORCH_CHECK in the launcher. Those raise, and this
+        # path is fail-closed by design, so each one has to be re-asserted here or an invariant
+        # maintained three files away turns into an exception mid-decode.
+        reqs = state_indices.shape[0]
+        qsl = attn_metadata.query_start_loc_d
+        num_accepted = attn_metadata.num_accepted_tokens
+        if num_accepted is None or num_accepted.numel() != reqs:
+            return False
+        if qsl.numel() != reqs + 1:
+            return False
+        if qsl.dtype != torch.int32 or num_accepted.dtype != torch.int32:
+            return False
+        if qsl.dim() != 1 or num_accepted.dim() != 1:
+            return False
+        # Slot 1+t is written for t < query_len <= state_indices.shape[1], and the taps sit at
+        # off/off+1 with off <= state_indices.shape[1]-1. Both need one slot more than the widest
+        # query.
+        return conv_state.shape[2] >= 1 + state_indices.shape[1]
+    except Exception:
+        return False
